\begin{frame}{Linguistic Motivation \& Mandated Pipeline}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Telugu Agglutinative Nature:}
      \begin{itemize}
        \item Highly inflectional Dravidian language.
        \item Words pack root, case markers, tense, aspect, mood, and agreement in a single token.
        \item Example: \textit{వెళ్ళాడు} (``he went'') $\rightarrow$ Root: \textit{వెళ్లు}, Past TAM, 3-Sg-M.
      \end{itemize}
      \vspace{0.3cm}
      \textbf{Linguistic Hierarchy Requirement:}
      \begin{itemize}
        \item Morphological features are \textbf{not independent}.
        \item Grammatical Agreement: Person/Number/Gender and Case logically depend on POS and Lexical Category.
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{The Mandated Hierarchical Flow}
        \centering
        \small
        \textbf{Input Sentence} $\rightarrow$ \textbf{MuRIL Encoder}\\
        $\Downarrow$\\
        \textbf{POS} Tagging\\
        $\Downarrow$\\
        \textbf{LexCat} (conditioned on POS)\\
        $\Downarrow$\\
        \textbf{Person} \& \textbf{Number} (conditioned on LexCat)\\
        $\Downarrow$\\
        \textbf{Gender} (conditioned on Person)\\
        $\Downarrow$\\
        \textbf{Case} (conditioned on LexCat + TAM representation)
      \end{block}
      \footnotesize
      \textit{*Chunk (CRF), Lemma (Seq2Seq Copy), and Suffix/TAM remain independent heads.}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% SLIDE 3: THE INITIAL CHALLENGE & DIAGNOSIS
% ------------------------------------------------------------------------------
\begin{frame}{Initial Implementation: The Cascade Failure}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{The Initial Hurdle ($\sim$60\% Drop)}
        \begin{itemize}
          \item First hierarchical implementation suffered severe performance degradation.
          \item POS \& LexCat remained high ($>$98\%), but downstream morphology collapsed:
          \begin{itemize}
            \item \textbf{Person accuracy:} dropped to $\sim$19\%
            \item \textbf{Case accuracy:} collapsed to $\sim$0.3\%
            \item \textbf{Number accuracy:} dropped to $\sim$60\%
          \end{itemize}
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.46\textwidth}
      \textbf{Root-Cause Analysis \& Diagnosis:}
      \begin{enumerate}
        \item \textbf{Hard Argmax Error Cascades:} Discrete predictions passed downstream propagated upstream errors irreversibly.
        \item \textbf{Representation Dilution:} Raw linear concatenation diluted 768-dim MuRIL representations with small dependency vectors.
        \item \textbf{The Destructive Masking Bug:} Validation logit masking defaulted to index 0 (First Person), destroying evaluation recall.
        \item \textbf{Gradient Poisoning by UNK:} Model learned shortcut paths classifying rare/missing gold labels as \texttt{[UNK]}.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% SLIDE 4: ARCHITECTURAL SOLUTIONS (THE V8 REDESIGN)
% ------------------------------------------------------------------------------
\begin{frame}{Core Breakthroughs: The V8 Redesign}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Soft Expectation Embeddings}
        \begin{itemize}
          \item Replaced hard \texttt{argmax} lookups with \textbf{differentiable expectation embeddings}:
          \[
            \mathbf{e}_{\text{soft}} = \sum_{c} P(y=c \mid \mathbf{x}) \cdot \mathbf{W}_{\text{emb}}[c]
          \]
          \item When uncertain, downstream heads receive a smooth probability mixture rather than an arbitrary wrong tag.
        \end{itemize}
      \end{block}

      \begin{block}{2. Confidence Gating \& Learned Trust}
        \begin{itemize}
          \item Learnable trust parameter $\alpha$ per connection:
          \[
            \text{Gate} = \sigma(\alpha) \cdot \max_c P(y=c)
          \]
          \item Downstream heads dynamically scale reliance on upstream outputs.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{3. Residual \texttt{FusionHead} Highway}
        \begin{itemize}
          \item Upstream dependency projected to $H=768$ and added as a \textbf{residual}:
          \[
            \mathbf{x}_{\text{fused}} = \mathbf{x}_{\text{MuRIL}} + \mathbf{W}_{\text{proj}}(\mathbf{e}_{\text{dep}})
          \]
          \item Stabilized with $\text{LayerNorm} \rightarrow \text{GELU} \rightarrow \text{Dropout}$.
          \item Ensures MuRIL contextual power is never diluted.
        \end{itemize}
      \end{block}

      \begin{block}{4. Multi-Task Uncertainty Loss}
        \begin{itemize}
          \item Kendall et al. homoscedastic variance weighting:
          \[
            \mathcal{L} = \sum_{t} \left(\frac{1}{2 e^{s_t}} \mathcal{L}_t + \frac{1}{2} s_t\right)
          \]
          \item Dynamically balances all 10 task losses.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% SLIDE 5: RIGOROUS UNK HANDLING STRATEGY
% ------------------------------------------------------------------------------
\begin{frame}{Robust Handling of Unknown Tokens (\texttt{[UNK]})}
  \begin{center}
    \small \textit{Handling corpus noise and missing annotations without corrupting learned representations}
  \end{center}

  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{1. Neutral Parameter Vector}
        \begin{itemize}
          \item Learned vector $\mathbf{v}_{\text{neutral}} \in \mathbb{R}^{64}$.
          \item If upstream predicts \texttt{[UNK]} or confidence $< 30\%$:
          \item Model smoothly substitutes $\mathbf{v}_{\text{neutral}}$.
          \item \textbf{Result:} Downstream heads rely on MuRIL without noisy bias.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{block}{2. Masked Class Loss}
        \begin{itemize}
          \item Real treebanks contain ambiguous gold \texttt{[UNK]} labels.
          \item Replaced gold \texttt{[UNK]} targets with index $-100$:
          \item Contributes \textbf{zero loss and zero gradient}.
          \item \textbf{Result:} Prevents model from learning \texttt{[UNK]} as a shortcut.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.32\textwidth}
      \begin{block}{3. Weight Damping}
        \begin{itemize}
          \item Standard inverse-frequency weighting inflates rare labels.
          \item Applied cap:
          \[
            w_{\text{UNK}} = \min(w_{\text{calc}}, 0.5)
          \]
          \item \textbf{Result:} Prevents rare unknown tags from dominating the gradient update.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% SLIDE 6: SEQUENCE DECODERS (LEMMA, SUFFIX & TAM)
% ------------------------------------------------------------------------------
\begin{frame}{Character-Level Sequence Decoders}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Lemma: Pointer-Generator Copy Mechanism}
        \begin{itemize}
          \item Telugu root stems share majority characters with input words.
          \item BiGRU Character Encoder + Bahdanau Attention.
          \item Computes generation probability $p_{\text{gen}}$:
          \[
            P(c) = p_{\text{gen}} P_{\text{vocab}}(c) + (1 - p_{\text{gen}}) \sum_{i: c_i = c} \alpha_i
          \]
          \item Enables direct character copy $\rightarrow$ \textbf{Exact Match: 96.38\%}.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Suffix \& TAM Decoders + Case Interlocking}
        \begin{itemize}
          \item Suffixes and TAM are abstract morphological tags.
          \item Pure generative GRU decoders (Exact Match: \textbf{98.14\%} \& \textbf{98.03\%}).
          \item \textbf{Dual Role of TAM Decoder:}
          \begin{enumerate}
            \item Generates character string output.
            \item Emits final 256-dim hidden state $\mathbf{h}_{\text{TAM}}$.
          \end{enumerate}
          \item $\mathbf{h}_{\text{TAM}}$ is concatenated with LexCat and fed directly to the \textbf{Case FusionHead}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
